#include <iostream>
using namespace std;

// Function to convert grade into points
float getGradePoint(char grade) {
    if (grade == 'A') return 10;
    else if (grade == 'B') return 8;
    else if (grade == 'C') return 6;
    else if (grade == 'D') return 4;
    else if (grade == 'F') return 0;
    else return 0;
}

int main() {
    int n;
    cout << "Enter number of courses: ";
    cin >> n;

    float totalCredits = 0;
    float totalGradePoints = 0;

    char grade;
    float credits;

    for (int i = 0; i < n; i++) {
        cout << "\nCourse " << i + 1 << endl;

        cout << "Enter grade (A/B/C/D/F): ";
        cin >> grade;

        cout << "Enter credit hours: ";
        cin >> credits;

        float gradePoint = getGradePoint(grade);

        totalCredits += credits;
        totalGradePoints += gradePoint * credits;
    }

    float gpa = totalGradePoints / totalCredits;

    cout << "\nTotal Credits: " << totalCredits;
    cout << "\nSemester GPA: " << gpa;

    // CGPA Calculation
    float prevCGPA, prevCredits;

    cout << "\n\nEnter previous CGPA: ";
    cin >> prevCGPA;

    cout << "Enter previous total credits: ";
    cin >> prevCredits;

    float finalCGPA = ((prevCGPA * prevCredits) + totalGradePoints) / (prevCredits + totalCredits);

    cout << "\nFinal CGPA: " << finalCGPA << endl;

    return 0;
}
